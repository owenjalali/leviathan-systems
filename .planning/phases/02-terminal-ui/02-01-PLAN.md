**Typewriter hook (inline in component or separate):**
```javascript
function useTypewriter(text, speed = 30) {
  const [displayedText, setDisplayedText] = useState('')
  const [isComplete, setIsComplete] = useState(false)

  useEffect(() => {
    if (!text) {
      setDisplayedText('')
      setIsComplete(false)
      return
    }

    let index = 0
    setDisplayedText('')
    setIsComplete(false)

    const interval = setInterval(() => {
      if (index < text.length) {
        setDisplayedText(text.slice(0, index + 1))
        index++
      } else {
        setIsComplete(true)
        clearInterval(interval)
      }
    }, speed)

    return () => clearInterval(interval)
  }, [text, speed])

  return { displayedText, isComplete }
}
```

**Staggered visibility logic:**
```javascript
const [visibleEvents, setVisibleEvents] = useState([])

// Show events when status becomes 'captured'
useEffect(() => {
  if (status === 'captured' && visibleEvents.length === 0) {
    // Stagger event appearance
    events.forEach((event, index) => {
      setTimeout(() => {
        setVisibleEvents(prev => [...prev, event.id])
      }, index * 400) // 400ms between each
    })
  }

  // Reset when going back to standby
  if (status === 'standby') {
    setVisibleEvents([])
  }
}, [status])
```
